  PENDING
  SUSPENDED

enum LocationScopeType {
  ALL
  SPECIFIC

enum RoleCategory {
  GENERAL
  CMS
  BILLING
  CUSTOM

// ----------------------------------------------------
// Staff & Collaborator RBAC
// ----------------------------------------------------

model Collaborator {
  id              String             @id @default(cuid())
  email           String             @unique
  name            String
  avatarUrl       String?
  status          CollaboratorStatus @default(ACTIVE)
  statusDate      DateTime           @default(now())
  locationType    LocationScopeType  @default(ALL)

  roles           CollaboratorRole[]
  locations       CollaboratorLocation[]

  createdAt       DateTime           @default(now())
  updatedAt       DateTime           @updatedAt

  @@index([status])
}

model Role {
  id              String             @id @default(cuid())
  slug            String             @unique
  name            String
  description     String
  category        RoleCategory       @default(CUSTOM)
  isSystemDefault Boolean            @default(false)

  permissions     RolePermission[]
  collaborators   CollaboratorRole[]

  createdAt       DateTime           @default(now())
  updatedAt       DateTime           @updatedAt

  @@index([category])
}

model Permission {
  id          String           @id @default(cuid())
  code        String           @unique // e.g. "roles.manage", "cms.posts.publish"
  name        String
  module      String           // e.g. "CMS", "BILLING", "SETTINGS"
  roles       RolePermission[]
}

model CollaboratorRole {
  collaboratorId String
  roleId         String
  assignedAt     DateTime     @default(now())

  collaborator   Collaborator @relation(fields: [collaboratorId], references: [id], onDelete: Cascade)
  role           Role         @relation(fields: [roleId], references: [id], onDelete: Cascade)

  @@id([collaboratorId, roleId])
}

model RolePermission {
  roleId       String
  permissionId String

  role         Role       @relation(fields: [roleId], references: [id], onDelete: Cascade)
  permission   Permission @relation(fields: [permissionId], references: [id], onDelete: Cascade)

  @@id([roleId, permissionId])
}

model CollaboratorLocation {
  collaboratorId String
  locationId     String

  collaborator   Collaborator @relation(fields: [collaboratorId], references: [id], onDelete: Cascade)

  @@id([collaboratorId, locationId])
}

// ----------------------------------------------------
// Member Portal Access (Gatekeeper)
// ----------------------------------------------------

model MemberRole {
  id          String               @id @default(cuid())
  name        String               @unique
  createdAt   DateTime             @default(now())
  updatedAt   DateTime             @updatedAt

  pageAccess  MemberRolePageAccess[]
}

model MemberPage {
  id          String               @id @default(cuid())
  slug        String               @unique
  title       String
  path        String

  roleAccess  MemberRolePageAccess[]
  createdAt   DateTime             @default(now())
}

model MemberRolePageAccess {
  memberRoleId String
  pageId       String

  memberRole   MemberRole @relation(fields: [memberRoleId], references: [id], onDelete: Cascade)
  page         MemberPage @relation(fields: [pageId], references: [id], onDelete: Cascade)

  @@id([memberRoleId, pageId])
}
