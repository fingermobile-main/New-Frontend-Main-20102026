2. Database Schema Design (PostgreSQL)
Below are the core relational structures required to prevent data structural gaps.
2.1 hallmarks_pages
CREATE TYPE hallmark_category AS ENUM ('PERSONAL', 'BUSINESS', 'GROUP', 'CREATOR');

CREATE TABLE hallmarks_pages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    owner_id UUID NOT NULL,
    display_name VARCHAR(30) NOT NULL UNIQUE,
    category hallmark_category NOT NULL,
    member_limit INT NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_name_length CHECK (char_length(display_name) BETWEEN 8 AND 30)
);

2.2 access_keys
CREATE TYPE access_key_type AS ENUM ('FOLLOWERSHIP', 'MEMBERSHIP');

CREATE TABLE access_keys (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    page_id UUID REFERENCES hallmarks_pages(id) ON DELETE CASCADE,
    key_token CHAR(16) UNIQUE NOT NULL,
    key_type access_key_type NOT NULL,
    expires_at TIMESTAMP WITH TIME ZONE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

2.3 coupons
CREATE TABLE coupons (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    page_id UUID REFERENCES hallmarks_pages(id) ON DELETE CASCADE,
    coupon_code VARCHAR(50) UNIQUE NOT NULL,
    is_redeemed BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

2.4 kyc_records
CREATE TYPE verification_status AS ENUM ('PENDING', 'APPROVED', 'REJECTED');

CREATE TABLE kyc_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL UNIQUE,
    full_legal_name VARCHAR(255) NOT NULL,
    is_above_22_years BOOLEAN NOT NULL,
    biometric_face_hash TEXT NOT NULL,
    status verification_status DEFAULT 'PENDING',
    reviewed_by UUID[],
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
